```julia
@inline function initial_condition_rayleigh_taylor(x, t, equations::CompressibleEulerEquations2D)
    gamma = equations.gamma
    
    # Density: Smooth transition from 2.0 (bottom) to 1.0 (top)
    rho = 2.0 + (1.0 - 2.0) * 0.5 * (1 + tanh(50 * (x[2] - 0.5)))
    
    # Pressure: Hydrostatic balance
    if x[2] < 0.5
        p = 2 * x[2] + 1
    else
        p = x[2] + 1.5
    end
    
    # Velocity: Perturbation
    c = sqrt(gamma * p / rho)
    v1 = 0.0
    v2 = -0.025 * c * cos(8 * pi * x[1]) * sin(pi * x[2])^6
    
    return prim2cons(SVector(rho, v1, v2, p), equations)
end

    # Energy source: rho * g * v = (rho * v) * g = rho_v2 * g
    return SVector(0.0, 0.0, g * rho, g * rho_v2)
end

    return u[1]
end
```

```julia
# 1. Find files and sort by time
files = readdir(npz_out, join=true)
filter!(x -> endswith(x, ".npz"), files)

# Parse times and sort
file_data = []
for f in files
    try
        t_str = split(basename(f), "_")[2]
        t_str = replace(t_str, ".npz" => "")
        t_val = parse(Float64, t_str)
        push!(file_data, (t=t_val, path=f))
    catch
        # Ignore malformed filenames
    end
end
sort!(file_data, by = x -> x.t)

# 2. Select targets
target_times = [0.0, 1.0, 2.0, 3.0]
selected_files = String[]

for target in target_times
    if isempty(file_data) break end
    # Find closest file
    diffs = [abs(x.t - target) for x in file_data]
    min_idx = argmin(diffs)
    push!(selected_files, file_data[min_idx].path)
end
```
