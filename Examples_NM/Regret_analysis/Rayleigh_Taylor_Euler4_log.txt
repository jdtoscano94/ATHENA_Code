@inline function initial_condition_rayleigh_taylor(x, t, equations::CompressibleEulerEquations2D)
    gamma = equations.gamma
    y = x[2]
    
    # Density: Smooth jump from 2.0 to 1.0 at y=0.5
    rho = 2.0 + (1.0 - 2.0) * 0.5 * (1 + tanh(50 * (y - 0.5)))
    
    # Pressure: Hydrostatic balance p_y = rho * g (with g=1 up)
    if y < 0.5
        p = 2 * y + 1
    else
        p = y + 1.5
    end
    
    # Velocity: Perturbation
    c = sqrt(gamma * p / rho)
    v1 = 0.0
    v2 = -0.025 * c * cos(8 * pi * x[1]) * sin(pi * y)^6
    
    return prim2cons(SVector(rho, v1, v2, p), equations)
end

@inline function source_terms_rayleigh_taylor(u, x, t, equations::CompressibleEulerEquations2D)
    g = 1.0 # Gravity in positive y-direction

    # Momentum source: rho * g
    # Energy source: rho * (g . v) = rho * (0*v1 + 1*v2) = rho * v2 = rho_v2
    return SVector(0.0, 0.0, g * rho, g * rho_v2)
end

    return Trixi.density(u, equations)
end
