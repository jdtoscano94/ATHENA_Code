    rho = 2.0 + (1.0 - 2.0) * 0.5 * (1 + tanh(50 * (y - 0.5)))

    if y < 0.5
        p = 2 * y + 1

function condition_save_npz(u, t, integrator)
    t_end = integrator.sol.prob.tspan[2]
    is_finished = t >= (t_end - 1e-5)
    
    # Check if time is close to a multiple of 0.1
    # Using a small tolerance to catch the time step
    dt_save = 0.1
    is_save_time = (abs(rem(t, dt_save)) < 1e-4) || (abs(rem(t, dt_save) - dt_save) < 1e-4)
    
    # Also save if it's the very first step (handled by init) or last step
    return is_save_time || is_finished
end
