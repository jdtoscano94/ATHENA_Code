@inline function initial_condition_rayleigh_taylor(x, t, equations::CompressibleEulerEquations2D)
    y = x[2]
    
    # Density: Smooth transition from 2.0 (bottom) to 1.0 (top)
    rho = 2.0 + (1.0 - 2.0) * 0.5 * (1.0 + tanh(50.0 * (y - 0.5)))
    
    # Pressure: Hydrostatic balance
    if y < 0.5
        p = 2.0 * y + 1.0
    else
        p = y + 1.5
    end
    
    # Velocity: Perturbation
    c = sqrt(equations.gamma * p / rho)
    u = 0.0
    v = -0.025 * c * cos(8.0 * pi * x[1]) * sin(pi * y)^6
    
    return prim2cons(SVector(rho, u, v, p), equations)
end

    # Energy source: rho * g * v (work done by gravity)
    return SVector(0.0, 0.0, g * rho, g * rho_v2)
end

end
