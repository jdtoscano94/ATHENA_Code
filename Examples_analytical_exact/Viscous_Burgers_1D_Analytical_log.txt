### Exact solution construction (periodic)

Introduce \(\phi(x,t)\) such that
\[
u(x,t) = -2\nu\,\partial_x \ln \phi(x,t) = -2\nu\,\frac{\phi_x}{\phi}.
\]
Then \(\phi\) satisfies the heat equation
\[
\partial_t \phi = \nu\,\partial_{xx}\phi.
\]

From \(u(x,0)=-\sin(\pi x)\),
\[
u(x,0) = -2\nu \frac{\phi_x(x,0)}{\phi(x,0)}
\quad\Rightarrow\quad
\phi(x,0)=\exp\!\left(-\frac{1}{2\nu}\int^x u(\xi,0)\,d\xi\right).
\]
Choose a periodic-consistent antiderivative:
\[
\int^x u(\xi,0)\,d\xi = \frac{\cos(\pi x)}{\pi}\quad(\text{constant irrelevant}),
\]
hence
\[
\phi(x,0)=\exp\!\left(-\frac{\cos(\pi x)}{2\nu\pi}\right).
\]

On a periodic domain of length \(2\), the heat solution can be written using a periodic heat kernel (image sum):
\[
\phi(x,t)=\int_{-1}^{1} \left[\sum_{m\in\mathbb{Z}}
\frac{1}{\sqrt{4\pi\nu t}}
\exp\!\left(-\frac{(x-y-2m)^2}{4\nu t}\right)\right]\phi(y,0)\,dy.
\]
Then recover
\[
u(x,t)=-2\nu\,\frac{\partial_x \phi(x,t)}{\phi(x,t)},
\]
with \(\partial_x \phi\) obtained by differentiating under the integral (analytic derivative of the kernel).

**Gauss–Hermite quadrature usage (as hinted):**  
Each Gaussian term in the kernel becomes a standard Gaussian-weight integral after the substitution
\[
x-y-2m = \sqrt{4\nu t}\,z,
\]
so integrals reduce to \(\int e^{-z^2} f(z)\,dz\), which is directly compatible with Gauss–Hermite nodes/weights. Truncate the image sum to the minimum \(m\) range needed for machine precision at given \((\nu,t)\).

"""
    gauss_hermite(n)

Compute the nodes and weights for Gauss-Hermite quadrature using the Golub-Welsch algorithm.
Integrals of the form ∫ exp(-x^2) f(x) dx are approximated by sum(w .* f(x)).
"""
function gauss_hermite(n)
    # Construct the symmetric tridiagonal matrix (Jacobi matrix)
    # For normalized Hermite polynomials, beta[i] = sqrt(i/2)
    i = 1:n-1
    beta = sqrt.(i ./ 2)
    alpha = zeros(n) # diagonal is zero for Hermite
    
    # Solve eigenvalue problem
    T = SymTridiagonal(alpha, beta)
    evals, evecs = eigen(T)
    
    # Nodes are eigenvalues
    x = evals
    
    # Weights are (sqrt(pi) * v[1]^2)
    w = sqrt(π) .* (evecs[1, :] .^ 2)
    
    return x, w
end

"""
    solve_viscous_burgers_exact(x_in, t, nu)

Computes the exact solution of the 1D Viscous Burgers' equation on a periodic domain [-1, 1]
using the Cole-Hopf transformation and Gauss-Hermite quadrature.

u(x,t) = -2ν * (∂x φ / φ)
φ(x,t) satisfies the heat equation.
"""
function solve_viscous_burgers_exact(x_in, t, nu)
    x = collect(Float64, x_in)
    
    # Handle initial condition (t=0)
    if t < 1e-9
        return -sin.(π .* x)
    end
    
    # Quadrature setup
    N_quad = 64
    z, w = gauss_hermite(N_quad)
    
    # Precompute constants
    sqrt_4nu_t = sqrt(4 * nu * t)
    inv_sqrt_nu_t = 1.0 / sqrt(nu * t)
    
    # Initial condition for phi: phi_0(y) = exp( -cos(πy) / (2πν) )
    # Derived from u0 = -sin(πx) -> ∫u0 = cos(πx)/π
    phi0_func(y) = exp(-cos(π * y) / (2 * π * nu))
    
    u_out = similar(x)
    
    # Image sum range for periodicity
    # Domain length L=2. Diffusion spread is small. m=-3:3 is sufficient for machine precision.
    m_range = -3:3
    
    for (i, xi) in enumerate(x)
        num_accum = 0.0
        den_accum = 0.0
        
        for m in m_range
            # Substitution: y = xi - 2m - sqrt(4νt)*z
            # We evaluate phi0 at y corresponding to the quadrature node z
            y_nodes = xi .- 2.0 * m .- sqrt_4nu_t .* z
            
            # Evaluate phi0
            phi_vals = phi0_func.(y_nodes)
            
            # Denominator integral: ∫ K * phi0 dy ~ ∑ w * phi0
            den_accum += dot(w, phi_vals)
            
            # Numerator integral: ∂x φ
            # ∂x K = K * (-z / sqrt(νt))
            # Integral becomes: ∑ w * phi0 * (-z / sqrt(νt))
            num_accum += dot(w, phi_vals .* (-z .* inv_sqrt_nu_t))
        end
        
        # u = -2ν * (φ_x / φ)
        u_out[i] = -2 * nu * (num_accum / den_accum)
    end
    
    return u_out
end

"""
    gauss_hermite(n)

Compute the nodes and weights for Gauss-Hermite quadrature using the Golub-Welsch algorithm.
Integrals of the form ∫ exp(-x^2) f(x) dx are approximated by sum(w .* f(x)).
"""
function gauss_hermite(n)
    # Construct the symmetric tridiagonal matrix (Jacobi matrix)
    # For normalized Hermite polynomials, beta[i] = sqrt(i/2)
    i = 1:n-1
    beta = sqrt.(i ./ 2)
    alpha = zeros(n) # diagonal is zero for Hermite
    
    # Solve eigenvalue problem
    T = SymTridiagonal(alpha, beta)
    evals, evecs = eigen(T)
    
    # Nodes are eigenvalues
    x = evals
    
    # Weights are (sqrt(pi) * v[1]^2)
    w = sqrt(π) .* (evecs[1, :] .^ 2)
    
    return x, w
end

"""
    solve_viscous_burgers_exact(x_in, t, nu)

Computes the exact solution of the 1D Viscous Burgers' equation on a periodic domain [-1, 1]
using the Cole-Hopf transformation and Gauss-Hermite quadrature.

u(x,t) = -2ν * (∂x φ / φ)
φ(x,t) satisfies the heat equation.
"""
function solve_viscous_burgers_exact(x_in, t, nu)
    x = collect(Float64, x_in)
    
    # Handle initial condition (t=0)
    if t < 1e-9
        return -sin.(π .* x)
    end
    
    # Quadrature setup
    N_quad = 64
    z, w = gauss_hermite(N_quad)
    
    # Precompute constants
    sqrt_4nu_t = sqrt(4 * nu * t)
    inv_sqrt_nu_t = 1.0 / sqrt(nu * t)
    
    # Initial condition for phi: phi_0(y) = exp( -cos(πy) / (2πν) )
    # Derived from u0 = -sin(πx) -> ∫u0 = cos(πx)/π
    phi0_func(y) = exp(-cos(π * y) / (2 * π * nu))
    
    u_out = similar(x)
    
    # Image sum range for periodicity
    # Domain length L=2. Diffusion spread is small. m=-3:3 is sufficient for machine precision.
    m_range = -3:3
    
    for (i, xi) in enumerate(x)
        num_accum = 0.0
        den_accum = 0.0
        
        for m in m_range
            # Substitution: y = xi - 2m - sqrt(4νt)*z
            # We evaluate phi0 at y corresponding to the quadrature node z
            y_nodes = xi .- 2.0 * m .- sqrt_4nu_t .* z
            
            # Evaluate phi0
            phi_vals = phi0_func.(y_nodes)
            
            # Denominator integral: ∫ K * phi0 dy ~ ∑ w * phi0
            den_accum += dot(w, phi_vals)
            
            # Numerator integral: ∂x φ
            # ∂x K = K * (-z / sqrt(νt))
            # Integral becomes: ∑ w * phi0 * (-z / sqrt(νt))
            num_accum += dot(w, phi_vals .* (-z .* inv_sqrt_nu_t))
        end
        
        # u = -2ν * (φ_x / φ)
        u_out[i] = -2 * nu * (num_accum / den_accum)
    end
    
    return u_out
end
