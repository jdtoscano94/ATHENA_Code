class Periodic_Fourier_Features(nn.Module):
    kmax: int = 1
    Lx: float = 2.0 
    @nn.compact
    def __call__(self, X: Array) -> Array:
        ks = jnp.arange(1, self.kmax + 1)
        Xper = 2 * jnp.pi * jnp.matmul(X, ks[None, :]) / self.Lx
        xcos = jnp.cos(Xper)
        xsin = jnp.sin(Xper)
        Xper = jnp.concatenate([xcos, xsin], axis=1)
        return Xper
```

    def Helmholtz(params, x, y, apply_fn):
        # Constants from the strategy
        k = 1.0
        a1 = 1.0
        a2 = 4.0
    
        # Model output
        u = apply_fn(params, x, y)
    
        # Source term g(x, y)
        g_xy = (k**2 - jnp.pi**2 * (a1**2 + a2**2)) * jnp.sin(jnp.pi * a1 * x) * jnp.sin(jnp.pi * a2 * y)
    
        # Compute uxx using forward-over-forward AD (jvp of jvp)
        f_for_x = lambda x_val: apply_fn(params, x_val, y)
        u_x_fn = lambda x_val: jvp(f_for_x, (x_val,), (jnp.ones_like(x_val),))[1]
        uxx = jvp(u_x_fn, (x,), (jnp.ones_like(x),))[1]
    
        # Compute uyy using forward-over-forward AD (jvp of jvp)
        f_for_y = lambda y_val: apply_fn(params, x, y_val)
        u_y_fn = lambda y_val: jvp(f_for_y, (y_val,), (jnp.ones_like(y_val),))[1]
        uyy = jvp(u_y_fn, (y,), (jnp.ones_like(y),))[1]
    
        # Helmholtz residual
        return uxx + uyy + k**2 * u - g_xy
    
    PDE_residuals = Helmholtz

    def SSBroyden_step(
        initial_weights, H0, key,
        xc, yc, lambdas,
        apply_fn, unflatten_func, batch_size, gamma, eta, k, c, static_options,it):
        key, subkey = jax.random.split(key)
        # 2. Update Lambdas ...
        params_it = unflatten_func(initial_weights)
        updated_lambdas = update_lambdas(apply_fn, params_it, lambdas, gamma, eta, xc, yc, it)
        _ , current_train_data_tuple = sample_points_pdf(
            subkey, batch_size, updated_lambdas, xc, yc, k, c
        )
        # 4. Prepare the full options dictionary for minimize
        current_options = dict(static_options)
        current_options['initial_H'] = H0
        # 5. Run the SSBroyden optimization for Nchange steps
        result = minimize(
            fun=loss_and_gradient,
            x0=initial_weights,
            args=(apply_fn, unflatten_func, *current_train_data_tuple),
            method='BFGS', # SSBroyden version of BFGs
            options=current_options)
        # 6. Recycle the Hessian purely
        new_H0 = result.hess_inv
        new_H0 = (new_H0 + jnp.transpose(new_H0)) / 2
        try:
            L = jnp.linalg.cholesky(new_H0)
            is_failed = jnp.any(jnp.isnan(L))
            final_H0 = jax.lax.cond(
                is_failed,
                lambda op: jnp.eye(op.shape[0], dtype=op.dtype),
                lambda op: op,
                operand=new_H0)
        except LinAlgError:
            final_H0 = jnp.eye(new_H0.shape[0], dtype=new_H0.dtype)
        return result.x, final_H0, key, result.fun, result.nit, updated_lambdas
    ```

class Periodic_Fourier_Features(nn.Module):
    kmax: int = 1
    Lx: float = 2.0 
    @nn.compact
    def __call__(self, X: Array) -> Array:
        ks = jnp.arange(1, self.kmax + 1)
        Xper = 2 * jnp.pi * jnp.matmul(X, ks[None, :]) / self.Lx
        xcos = jnp.cos(Xper)
        xsin = jnp.sin(Xper)
        Xper = jnp.concatenate([xcos, xsin], axis=1)
        return Xper
```

class Periodic_Fourier_Features(nn.Module):
    kmax: int = 1
    Lx: float = 2.0 
    @nn.compact
    def __call__(self, X: Array) -> Array:

class Periodic_Fourier_Features(nn.Module):
    kmax: int = 1
    Lx: float = 2.0 
    @nn.compact
    def __call__(self, X: Array) -> Array:
        ks = jnp.arange(1, self.kmax + 1)
        Xper = 2 * jnp.pi * jnp.matmul(X, ks[None, :]) / self.Lx
        xcos = jnp.cos(Xper)
        xsin = jnp.sin(Xper)
        Xper = jnp.concatenate([xcos, xsin], axis=1)
        return Xper
```
