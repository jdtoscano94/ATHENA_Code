function condition_save_npz(u, t, integrator)
    dt_save = 0.1
    t_end = integrator.sol.prob.tspan[2]
    
    # Check if we just passed a save point
    is_save_time = (t - integrator.tprev) > 0 && (floor(t / dt_save) > floor(integrator.tprev / dt_save))
    is_finished = t >= (t_end - 1e-5)

end
